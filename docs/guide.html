Project 01 RAG over your own documents

⏱ 4–8 hours💶 < $1 LLM credit; embeddings free on CPU🧰 IDE or Colab, no GPU

Goal. Ask a question about a document collection and get an answer that cites numbered sources — document and page — and says “I don't know” when the documents don't contain the answer. Then prove it works with an evaluation that prints a pass rate.

PDFs / HF dataset→chunk 300–500 tokens, ~50 overlap→embed (bge-small)→Chroma→top-k retrieve→prompt with [1][2]→LLM via OpenRouter→eval

ingest.py — load, chunk, embed, store

Reads the Hugging Face dataset rag-datasets/rag-mini-wikipedia (3,200 Wikipedia passages + 918 Q&A pairs, config text-corpus, split passages) and/or your own PDFs page by page with pypdf, so every chunk remembers source and page.
ingest.py
"""Step 1-4: load -> chunk -> embed -> index (Chroma, persisted to disk).

Usage:
  python ingest.py --hf --limit 3200      # rag-mini-wikipedia passages
  python ingest.py --pdfs my_docs/        # your own PDFs (one record per page)
  python ingest.py --hf --pdfs my_docs/   # both
"""
import argparse
import pathlib

import chromadb
from sentence_transformers import SentenceTransformer

import config


def load_hf(limit: int | None):
    from datasets import load_dataset
    ds = load_dataset("rag-datasets/rag-mini-wikipedia", "text-corpus", split="passages")
    if limit:
        ds = ds.select(range(min(limit, len(ds))))
    for row in ds:
        # each passage is one "page" of a pseudo-document called wiki-<id>
        yield {"text": row["passage"], "source": f"wiki-{row['id']}", "page": 1}


def load_pdfs(folder: str):
    from pypdf import PdfReader
    for pdf in sorted(pathlib.Path(folder).glob("**/*.pdf")):
        reader = PdfReader(str(pdf))
        for i, page in enumerate(reader.pages, start=1):
            text = page.extract_text() or ""
            if text.strip():
                yield {"text": text, "source": pdf.name, "page": i}


def chunk(text: str, tokenizer, size: int, overlap: int) -> list[str]:
    """Token-based sliding window so 'size' really means model tokens.
    We slice the ORIGINAL text via character offsets, so case and spacing survive."""
    enc = tokenizer(text, add_special_tokens=False, return_offsets_mapping=True)
    offsets = enc["offset_mapping"]
    if not offsets:
        return []
    step = max(1, size - overlap)
    out = []
    for start in range(0, len(offsets), step):
        window = offsets[start:start + size]
        out.append(text[window[0][0]:window[-1][1]].strip())
        if start + size >= len(offsets):
            break
    return out


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--hf", action="store_true", help="index rag-mini-wikipedia")
    ap.add_argument("--limit", type=int, default=None, help="max HF passages")
    ap.add_argument("--pdfs", default=None, help="folder with your PDFs")
    ap.add_argument("--reset", action="store_true", help="drop the collection first")
    args = ap.parse_args()
    if not (args.hf or args.pdfs):
        ap.error("pass --hf and/or --pdfs FOLDER")

    model = SentenceTransformer(config.EMBED_MODEL)
    tok = model.tokenizer

    client = chromadb.PersistentClient(path=config.DB_DIR)
    if args.reset:
        try:
            client.delete_collection(config.COLLECTION)
        except Exception:
            pass
    col = client.get_or_create_collection(config.COLLECTION, metadata={"hnsw:space": "cosine"})

    records = []
    if args.hf:
        records += list(load_hf(args.limit))
    if args.pdfs:
        records += list(load_pdfs(args.pdfs))
    print(f"loaded {len(records)} pages/passages")

    ids, docs, metas = [], [], []
    for r in records:
        for j, c in enumerate(chunk(r["text"], tok, config.CHUNK_TOKENS, config.CHUNK_OVERLAP)):
            ids.append(f"{r['source']}::p{r['page']}::c{j}")
            docs.append(c)
            metas.append({"source": r["source"], "page": r["page"], "chunk": j})
    print(f"made {len(docs)} chunks (size={config.CHUNK_TOKENS}, overlap={config.CHUNK_OVERLAP})")

    B = 256
    for i in range(0, len(docs), B):
        vecs = model.encode(docs[i:i + B], normalize_embeddings=True, show_progress_bar=False)
        col.upsert(ids=ids[i:i + B], documents=docs[i:i + B],
                   metadatas=metas[i:i + B], embeddings=vecs.tolist())
        print(f"  indexed {min(i + B, len(docs))}/{len(docs)}")
    print(f"done. collection '{config.COLLECTION}' has {col.count()} chunks in ./{config.DB_DIR}")


if __name__ == "__main__":
    main()

rag.py — retrieve, build the cited prompt, answer

Two “I don't know” guards: (1) if even the best chunk is too far away (cosine distance > 0.6), we return IDK without calling the LLM (cheaper, no hallucination); (2) the system prompt orders the model to answer only from the passages and reply IDK otherwise.
rag.py
"""Step 5-6: retrieve top-k chunks and answer with numbered citations.

Usage:
  python rag.py "Who was the sixteenth president of the United States?"
  python rag.py --no-llm "question"     # retrieval only, no API key needed
"""
import argparse
import os
from functools import lru_cache

import chromadb
from openai import OpenAI
from sentence_transformers import SentenceTransformer

import config

IDK = "I don't know based on the provided documents."

SYSTEM = (
    "You answer questions using ONLY the numbered context passages. "
    "After every sentence that uses a passage, cite it like [1] or [2][3]. "
    f"If the passages do not contain the answer, reply exactly: {IDK} "
    "Never use outside knowledge. Be concise."
)


@lru_cache
def _model():
    return SentenceTransformer(config.EMBED_MODEL)


@lru_cache
def _collection():
    client = chromadb.PersistentClient(path=config.DB_DIR)
    return client.get_collection(config.COLLECTION)


@lru_cache
def _llm():
    key = os.getenv("OPENROUTER_API_KEY")
    if not key:
        raise SystemExit("OPENROUTER_API_KEY missing: put it in .env")
    return OpenAI(base_url=config.OPENROUTER_BASE_URL, api_key=key)


def retrieve(question: str, k: int = config.TOP_K) -> list[dict]:
    q = _model().encode([question], normalize_embeddings=True).tolist()
    res = _collection().query(query_embeddings=q, n_results=k,
                              include=["documents", "metadatas", "distances"])
    hits = []
    for doc, meta, dist in zip(res["documents"][0], res["metadatas"][0], res["distances"][0]):
        hits.append({"text": doc, "source": meta["source"], "page": meta["page"], "distance": dist})
    return hits


def build_prompt(question: str, hits: list[dict]) -> str:
    blocks = [f"[{i}] (source: {h['source']}, page {h['page']})\n{h['text']}"
              for i, h in enumerate(hits, start=1)]
    return "Context passages:\n\n" + "\n\n".join(blocks) + f"\n\nQuestion: {question}\nAnswer:"


def answer(question: str, k: int = config.TOP_K, use_llm: bool = True) -> dict:
    hits = [h for h in retrieve(question, k) if h["distance"] <= config.MAX_DISTANCE]
    if not hits:  # guard 1: nothing relevant retrieved -> refuse without calling the LLM
        return {"answer": IDK, "sources": []}
    if not use_llm:
        return {"answer": "(LLM skipped)", "sources": hits}
    resp = _llm().chat.completions.create(
        model=config.LLM_MODEL,
        temperature=0,
        messages=[{"role": "system", "content": SYSTEM},
                  {"role": "user", "content": build_prompt(question, hits)}],
    )
    text = resp.choices[0].message.content.strip()  # guard 2: the system prompt
    return {"answer": text, "sources": hits, "usage": resp.usage}


def format_sources(hits: list[dict]) -> str:
    return "\n".join(f"  [{i}] {h['source']} p.{h['page']}  (distance {h['distance']:.3f})"
                     for i, h in enumerate(hits, start=1))


if __name__ == "__main__":
    ap = argparse.ArgumentParser()
    ap.add_argument("question")
    ap.add_argument("--no-llm", action="store_true")
    a = ap.parse_args()
    out = answer(a.question, use_llm=not a.no_llm)
    print(out["answer"])
    if out["sources"]:
        print("\nSources:\n" + format_sources(out["sources"]))

def load_tests(n: int, seed: int = 42, skip_yes_no: bool = False):
    ds = load_dataset("rag-datasets/rag-mini-wikipedia", "question-answer", split="test")
    rows = list(ds)
    if skip_yes_no:  # "yes"/"no" never appears verbatim in a passage
        rows = [r for r in rows if r["answer"].strip().lower().rstrip(".") not in ("yes", "no")]
    random.Random(seed).shuffle(rows)
    return rows[:n]


def regex_for(expected: str) -> str:
    # whole-word, case-insensitive match of the reference answer
    return r"\b" + re.escape(expected.strip().rstrip(".")) + r"\b"


def judge(q, ref, a, hits) -> dict:
    ctx = "\n".join(f"[{i}] {h['text'][:800]}" for i, h in enumerate(hits, 1))
    resp = rag._llm().chat.completions.create(
        model=config.LLM_MODEL, temperature=0,
        response_format={"type": "json_object"},
        messages=[{"role": "user", "content": JUDGE_PROMPT.format(q=q, ref=ref, a=a, ctx=ctx)}],
    )
    try:
        return json.loads(resp.choices[0].message.content)
    except json.JSONDecodeError:
        return {"correct": False, "grounded": False, "reason": "judge returned bad JSON"}


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--n", type=int, default=50)
    ap.add_argument("--judge", action="store_true")
    ap.add_argument("--retrieval", action="store_true",
                    help="only check whether the answer text appears in the top-k chunks")
    args = ap.parse_args()

    tests = load_tests(args.n, skip_yes_no=args.retrieval)
    passed = judged_ok = 0
    for t in tests:
        q, ref = t["question"], t["answer"]
        if args.retrieval:
            hits = rag.retrieve(q)
            ok = any(re.search(regex_for(ref), h["text"], re.I) for h in hits)
            passed += ok
            continue
        out = rag.answer(q)
        a = out["answer"]
        cited = re.search(r"\[\d+\]", a)
        correct = re.search(regex_for(ref), a, re.I)
        ok = bool(cited and correct)
        passed += ok
        line = f"{'PASS' if ok else 'FAIL'} | {q[:60]:60} | ref={ref[:20]!r}"
        if args.judge and out["sources"]:
            j = judge(q, ref, a, out["sources"])
            judged_ok += bool(j.get("correct") and j.get("grounded"))
            line += f" | judge={j.get('correct')}/{j.get('grounded')}"
        print(line)

    label = "retrieval hit rate" if args.retrieval else "pass rate"
    print(f"\n{label}: {passed / len(tests):.0%}  ({passed}/{len(tests)})")
    if args.judge:
        print(f"judge pass rate (correct AND grounded): {judged_ok / len(tests):.0%}")


if __name__ == "__main__":
    main()

"""Run untrusted, model-written Python somewhere that is NOT your laptop."""
import os
import pathlib
import subprocess
import tempfile


def run_in_e2b(code: str, timeout: int = 60) -> str:
    from e2b_code_interpreter import Sandbox  # needs E2B_API_KEY in the environment
    # isolated microVM; lifetime capped at 5 min, no internet from inside
    sbx = Sandbox.create(timeout=300, allow_internet_access=False)
    try:
        ex = sbx.run_code(code, timeout=timeout)
        out = "".join(ex.logs.stdout) + "".join(ex.logs.stderr)
        if ex.error:
            out += f"\nERROR {ex.error.name}: {ex.error.value}"
        return out[-4000:] or "(no output)"
    finally:
        sbx.kill()                            # always destroy the VM


def run_in_docker(code: str, timeout: int = 60) -> str:
    """Throwaway container: no network, 512 MB RAM, 1 CPU, read-only code mount."""
    with tempfile.TemporaryDirectory() as d:
        pathlib.Path(d, "main.py").write_text(code)
        cmd = ["docker", "run", "--rm", "--network", "none", "--memory", "512m",
               "--cpus", "1", "--pids-limit", "128", "-v", f"{d}:/work:ro",
               "python:3.11-slim", "python", "/work/main.py"]
        try:
            p = subprocess.run(cmd, capture_output=True, text=True, timeout=timeout)
        except subprocess.TimeoutExpired:
            return "ERROR: timed out"
        return ((p.stdout + p.stderr)[-4000:]) or "(no output)"


def run_code(code: str) -> str:
    backend = os.getenv("SANDBOX", "e2b").lower()
    return run_in_docker(code) if backend == "docker" else run_in_e2b(code)

"""Long-horizon agent: plan -> parallel sub-agents -> writer, with guardrails.

  python agent.py "Research what HNSW is and write report.md with a small Python benchmark"
"""
import asyncio
import os
import pathlib
import re
import sys
from decimal import Decimal

from dotenv import load_dotenv
from pydantic import BaseModel, Field
from pydantic_ai import (Agent, ApprovalRequired, ModelRetry, DeferredToolRequests, DeferredToolResults, RunContext,
                         Tool, ToolDenied, UsageLimitExceeded, UsageLimits)
from pydantic_ai.capabilities import HandleDeferredToolCalls
from pydantic_ai.messages import ModelResponse, ToolCallPart, ToolReturnPart

from logger import StepLogger
from sandbox import run_code

load_dotenv()
os.environ.setdefault("PYDANTIC_AI_NO_BANNER", "1")
WORKSPACE = pathlib.Path("workspace").resolve()
WORKSPACE.mkdir(exist_ok=True)
MAX_STEPS = int(os.getenv("MAX_STEPS", "25"))
MAX_COST = Decimal(os.getenv("MAX_COST_USD", "0.50"))
ALLOWED_TOOLS = {"search_files", "read_file", "write_file", "run_python"}  # allowlist


def get_model():
    name = os.getenv("AGENT_MODEL", "openai/gpt-4o-mini")
    if name == "test":                       # offline dry run, no API key, no cost
        from pydantic_ai.models.test import TestModel
        return TestModel()
    from pydantic_ai.models.openrouter import OpenRouterModel
    return OpenRouterModel(name)             # reads OPENROUTER_API_KEY


# ---------- typed tools (Pydantic validates every argument) ----------
def _safe(path: str) -> pathlib.Path:
    p = (WORKSPACE / path).resolve()
    if WORKSPACE not in p.parents and p != WORKSPACE:
        raise ValueError("path escapes the workspace")   # guardrail: no ../../etc
    return p


def search_files(query: str) -> list[str]:
    """Search the workspace for files whose text contains the query (case-insensitive)."""
    hits = []
    for f in WORKSPACE.rglob("*"):
        if f.is_file() and query.lower() in f.read_text(errors="ignore").lower():
            hits.append(str(f.relative_to(WORKSPACE)))
    return hits[:20] or ["no matches"]


def read_file(path: str) -> str:
    """Read a text file from the workspace (max 8000 characters)."""
    p = _safe(path)
    if not p.is_file():
        raise ModelRetry(f"{path} does not exist; call search_files first")  # model retries
    return p.read_text(errors="ignore")[:8000]


def write_file(ctx: RunContext, path: str, content: str) -> str:
    """Create or overwrite a text file in the workspace."""
    p = _safe(path)
    if p.exists() and not ctx.tool_call_approved:
        raise ApprovalRequired          # overwriting = risky -> ask a human first
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text(content)
    return f"wrote {len(content)} chars to {path}"


def run_python(code: str) -> str:
    """Execute Python code in an isolated sandbox (E2B or Docker) and return stdout/stderr."""
    if re.search(r"\b(rm -rf|shutil\.rmtree|os\.remove)\b", code):
        return "blocked: destructive operation"          # cheap static guardrail
    try:
        return run_code(code)
    except Exception as e:                               # report, don't crash the run
        return f"sandbox error: {type(e).__name__}: {e}"


def _tools():
    all_tools = [Tool(search_files), Tool(read_file), Tool(write_file), Tool(run_python)]
    return [t for t in all_tools if t.name in ALLOWED_TOOLS]


async def ask_human(ctx: RunContext, requests: DeferredToolRequests) -> DeferredToolResults:
    approvals = {}
    for call in requests.approvals:
        if os.getenv("AUTO_APPROVE") == "1":
            approvals[call.tool_call_id] = True
            continue
        ans = await asyncio.to_thread(input, f"\nApprove {call.tool_name}({call.args})? [y/N] ")
        approvals[call.tool_call_id] = True if ans.strip().lower() == "y" else ToolDenied("Human said no")
    return requests.build_results(approvals=approvals)


# ---------- agents ----------
class Plan(BaseModel):
    subtasks: list[str] = Field(min_length=1, max_length=4,
                                description="independent subtasks that can run in parallel")


planner = Agent(get_model(), output_type=Plan,
                instructions="Split the task into 1-4 independent subtasks. Do not do them.")

worker = Agent(get_model(), tools=_tools(),
               capabilities=[HandleDeferredToolCalls(handler=ask_human)],
               instructions=("You complete ONE subtask using tools. Save findings to "
                             "workspace files named notes_<topic>.md. Use run_python to verify "
                             "any code. Finish with a 3-line summary."))

writer = Agent(get_model(), tools=_tools(),
               capabilities=[HandleDeferredToolCalls(handler=ask_human)],
               instructions="Read the notes_*.md files and write the final deliverable requested.")


async def logged_run(agent: Agent, prompt: str, log: StepLogger, name: str):
    """agent.iter() lets us see every model response and tool result as it happens."""
    limits = UsageLimits(request_limit=MAX_STEPS, tool_calls_limit=MAX_STEPS * 2,
                         cost_limit=MAX_COST)
    async with agent.iter(prompt, usage_limits=limits) as run:
        async for node in run:
            if Agent.is_call_tools_node(node):
                resp: ModelResponse = node.model_response
                u = resp.usage
                log.log(kind="model", agent=name, model=resp.model_name,
                        input_tokens=u.input_tokens, output_tokens=u.output_tokens,
                        cost_usd=getattr(u, "cost", None))
                for part in resp.parts:
                    if isinstance(part, ToolCallPart):
                        log.log(kind="call", agent=name, tool=part.tool_name, input=part.args)
            elif Agent.is_model_request_node(node):
                for part in node.request.parts:
                    if isinstance(part, ToolReturnPart):
                        log.log(kind="result", agent=name, tool=part.tool_name,
                                output=str(part.content)[:500])
        return run.result.output, run.usage


async def main(task: str):
    log = StepLogger("run")
    plan, _ = await logged_run(planner, task, log, "planner")
    print("plan:", plan.subtasks)
    # sub-agents in parallel; one failure doesn't kill the others
    results = await asyncio.gather(
        *[logged_run(worker, f"Overall task: {task}\nYour subtask: {s}", log, f"worker{i}")
          for i, s in enumerate(plan.subtasks)],
        return_exceptions=True)
    for s, r in zip(plan.subtasks, results):
        status = "ERROR " + repr(r) if isinstance(r, Exception) else "ok"
        print(f"- {s[:60]}: {status}")
    final, usage = await logged_run(writer, f"Task: {task}", log, "writer")
    print("\nFINAL:\n", final)
    print(f"\nlog file: {log.path}")


if __name__ == "__main__":
    try:
        asyncio.run(main(" ".join(sys.argv[1:]) or "Write hello.md that says hello"))
    except UsageLimitExceeded as e:
        print("STOPPED by budget/step guardrail:", e)
How the pieces map to Pydantic AI 2.x: Tool(fn) turns a typed Python function into a tool (the type hints + docstring become the JSON schema the model sees). output_type=Plan forces a validated Pydantic object. raise ModelRetry(...) sends an error back to the model to try again. raise ApprovalRequired pauses the tool call; the HandleDeferredToolCalls capability calls ask_human, which returns approvals or ToolDenied. agent.iter() lets us see every node to log it.

"""Read vLLM's Prometheus /metrics and print server-side averages.

  python metrics_summary.py --url http://localhost:8000
Histogram metrics expose <name>_sum and <name>_count; average = sum / count.
"""
import argparse
import re

import httpx

HISTOGRAMS = {
    "vllm:time_to_first_token_seconds": "TTFT",
    "vllm:inter_token_latency_seconds": "inter-token latency",
    "vllm:e2e_request_latency_seconds": "end-to-end latency",
    "vllm:request_queue_time_seconds": "queue time",
}
COUNTERS = {"vllm:prompt_tokens_total": "prompt tokens",
            "vllm:generation_tokens_total": "generated tokens",
            "vllm:request_success_total": "finished requests"}
GAUGES = {"vllm:num_requests_running": "running now", "vllm:num_requests_waiting": "waiting now",
          "vllm:kv_cache_usage_perc": "KV cache used (0-1)"}


def total(text: str, name: str) -> float:
    """Sum a metric across all label sets, e.g. vllm:x_sum{model_name="..."} 1.23"""
    pat = re.compile(rf"^{re.escape(name)}(?:{{[^}}]*}})?\s+([0-9.eE+-]+)$", re.M)
    return sum(float(v) for v in pat.findall(text))


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--url", default="http://localhost:8000")
    text = httpx.get(f"{ap.parse_args().url}/metrics", timeout=10).text
    for name, label in HISTOGRAMS.items():
        s, c = total(text, name + "_sum"), total(text, name + "_count")
        print(f"{label:>22}: " + (f"avg {1000 * s / c:.1f} ms over {int(c)} samples" if c else "no data yet"))
    for name, label in {**COUNTERS, **GAUGES}.items():
        print(f"{label:>22}: {total(text, name):g}")


if __name__ == "__main__":
    main()

load_test.py — what users feel at 1, 4, 16 concurrent users

Client-side measurement over streaming /v1/chat/completions. Works against vLLM, SGLang or any OpenAI-compatible server.
load_test.py
"""Async load test for any OpenAI-compatible server (vLLM, SGLang, ...).
Measures, per request, from the client side:
  TTFT  = time to first streamed token
  ITL   = gaps between streamed chunks (inter-token latency)
  and overall output tokens/second (throughput).

  python load_test.py --url http://localhost:8000 --model unsloth/Llama-3.2-1B-Instruct \
      --concurrency 1 4 16 --requests 32 --max-tokens 128
"""
import argparse
import asyncio
import json
import statistics
import time

import httpx

PROMPT = "Explain retrieval-augmented generation to a CFO in about 150 words."


def pct(xs, p):
    xs = sorted(xs)
    return xs[min(len(xs) - 1, int(round(p / 100 * (len(xs) - 1))))] if xs else float("nan")


async def one_request(client, url, model, max_tokens):
    body = {"model": model, "stream": True, "max_tokens": max_tokens, "temperature": 0,
            "messages": [{"role": "user", "content": PROMPT}],
            "stream_options": {"include_usage": True}}
    t0 = time.perf_counter()
    ttft, last, gaps, chunks, usage_tokens = None, None, [], 0, None
    async with client.stream("POST", f"{url}/v1/chat/completions", json=body) as r:
        r.raise_for_status()
        async for line in r.aiter_lines():
            if not line.startswith("data: "):
                continue
            data = line[6:]
            if data.strip() == "[DONE]":
                break
            msg = json.loads(data)
            if msg.get("usage"):
                usage_tokens = msg["usage"].get("completion_tokens")
            if not msg.get("choices") or not msg["choices"][0]["delta"].get("content"):
                continue
            now = time.perf_counter()
            if ttft is None:
                ttft = now - t0
            else:
                gaps.append(now - last)
            last = now
            chunks += 1
    return {"ttft": ttft, "gaps": gaps, "e2e": time.perf_counter() - t0,
            "tokens": usage_tokens or chunks}


async def run_level(url, model, concurrency, n, max_tokens):
    sem = asyncio.Semaphore(concurrency)
    async with httpx.AsyncClient(timeout=300) as client:
        async def guarded():
            async with sem:
                return await one_request(client, url, model, max_tokens)
        t0 = time.perf_counter()
        res = await asyncio.gather(*[guarded() for _ in range(n)], return_exceptions=True)
        wall = time.perf_counter() - t0
    ok = [r for r in res if isinstance(r, dict) and r["ttft"] is not None]
    ttfts = [r["ttft"] * 1000 for r in ok]
    itls = [g * 1000 for r in ok for g in r["gaps"]]
    toks = sum(r["tokens"] for r in ok)
    return {"concurrency": concurrency, "ok": len(ok), "failed": n - len(ok),
            "ttft_p50_ms": statistics.median(ttfts) if ttfts else float("nan"),
            "ttft_p95_ms": pct(ttfts, 95),
            "itl_p50_ms": statistics.median(itls) if itls else float("nan"),
            "itl_p95_ms": pct(itls, 95),
            "throughput_tok_s": toks / wall, "req_s": len(ok) / wall}


async def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--url", default="http://localhost:8000")
    ap.add_argument("--model", default="unsloth/Llama-3.2-1B-Instruct")
    ap.add_argument("--concurrency", type=int, nargs="+", default=[1, 4, 16])
    ap.add_argument("--requests", type=int, default=32)
    ap.add_argument("--max-tokens", type=int, default=128)
    a = ap.parse_args()
    rows = [await run_level(a.url, a.model, c, a.requests, a.max_tokens) for c in a.concurrency]
    print("| concurrency | ok/failed | TTFT p50 (ms) | TTFT p95 (ms) | ITL p50 (ms) | ITL p95 (ms) | output tok/s | req/s |")
    print("|---|---|---|---|---|---|---|---|")
    for r in rows:
        print(f"| {r['concurrency']} | {r['ok']}/{r['failed']} | {r['ttft_p50_ms']:.0f} | {r['ttft_p95_ms']:.0f} "
              f"| {r['itl_p50_ms']:.1f} | {r['itl_p95_ms']:.1f} | {r['throughput_tok_s']:.0f} | {r['req_s']:.2f} |")


if __name__ == "__main__":
    asyncio.run(main())

Traps

⚠ Too-small eval set
20 questions → one answer = 5 pp. Differences under ~10 pp are noise.
⚠ Overlap with tuning data
If the same questions chose k and chunk size, the score is optimistic. Ask for a separate dev set.
⚠ LLM-judge bias
Judges favour long, confident answers and their own model family. Use a different model, a rubric, and a human spot-check.
⚠ Averages hiding failure clusters
Break down by question type, document, language. One category at 20% can sink adoption.
⚠ Non-deterministic runs reported once
Ask for ≥3 runs and the spread; temperature 0 doesn't make APIs fully deterministic.
